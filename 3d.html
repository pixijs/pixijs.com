import { Assets } from 'pixi.js';
import {
  DirectionalLight, Model3D, View3D,
} from '@pixi/3d';


const view = new View3D({ autoResize: true });
app.stage.addChild(view);


// glTF loads through Assets, like a texture
const fox = new Model3D({
  source: await Assets.load('fox.glb'),
  scale: 0.01,
});
// and its animations play by name
fox.player?.play('Survey');


const sun = new DirectionalLight({
  castShadow: true,
});
view.root.addChild(fox, sun);

import {
  ContainerTexture, CubeGeometry,
  Material3D, Mesh3D,
} from '@pixi/3d';


// screen is any 2D container: sprites, text, UI.
// It's drawn into a texture every frame
const face = new ContainerTexture({
  container: screen,
  renderer: app.renderer,
  autoUpdate: true,
});


// and worn by a cube
const geometry = new CubeGeometry();
const material = new Material3D({
  baseColorTexture: face,
});
const cube = new Mesh3D({ geometry, material });
view.root.addChild(cube);

import { Assets } from 'pixi.js';
import { Sprite3D } from '@pixi/3d';
import { OrbitCamera } from '@pixi/3d/extras';


const texture = await Assets.load('bunny.png');


// textured quads that turn to face the camera
for (let i = 0; i < 60; i++) {
  const angle = (i / 60) * Math.PI * 2;
  const bunny = new Sprite3D({
    texture,
    billboardMode: 'spherical',
    height: 0.4,
  });


  bunny.x = Math.cos(angle) * 2.5;
  bunny.z = Math.sin(angle) * 2.5;
  view.root.addChild(bunny);
}
new OrbitCamera({ view, autoRotate: true });

import {
  CubeGeometry, Material3D, Mesh3D,
} from '@pixi/3d';


const geometry = new CubeGeometry();
const material = new Material3D();


for (let i = 0; i < 15; i++) {
  const cube = new Mesh3D({ geometry, material });


  cube.x = (i % 5) * 1.5 - 3;
  cube.z = Math.floor(i / 5) * 1.5 - 1.5;
  // the events you use in 2D, picked with a ray
  cube.eventMode = 'static';
  cube.cursor = 'pointer';
  cube.on('pointerover', () => (cube.y = 0.4));
  cube.on('pointerout', () => (cube.y = 0));
  cube.on('pointertap', () => cube.rotation.y++);
  view.root.addChild(cube);
}

FEATURES

A full 3D engine.

  • glTF 2.0

    Skins, morph targets and animations, KHR materials and lights, Draco and meshopt compression. Load with Assets and play animations with model.player.

  • Lights & shadows

    Directional, point, spot and ambient lights, up to 32 per view, with soft PCF shadows and cascades.

  • Post-processing

    Bloom, SSAO, FXAA, tilt-shift and outlines, from @pixi/3d/extras.

  • PBR materials

    Metallic-roughness with clearcoat, sheen, iridescence, transmission, specular and anisotropy, lit by environment maps or a hemisphere.

  • Particles

    Fire, smoke, sparks and flipbooks from one options object, with ParticleSystem3D.

  • Instancing

    BatchedMesh3D draws thousands of copies, culled on the GPU under WebGPU.

  • Cameras

    OrbitCamera and FPSCamera, ready to drop in.

  • Stylised

    Toon, Phong and flat materials for when PBR is too real.

  • Lean

    Import @pixi/3d and it downloads only the backend your renderer picks.

NOTHING NEW TO LEARN

The same easy-to-use API.

  1. 01View3D

    Same stage

    A View3D is a container, so you can put it under sprites, over a background or behind a 2D filter.

    const view = new View3D({ autoResize: true });
app.stage.addChild(background, view, hud);

  2. 02Model3D

    Same loader

    Assets loads glTF and GLB, including Draco and meshopt compression.

    const robot = new Model3D(
  await Assets.load<Scene3DSource>('robot.glb'),
);
view.root.addChild(robot);

  3. 03eventMode

    Same events

    Ray-picked pointer events with the names, bubbling and cursors you use in 2D.

    mesh.eventMode = 'static';
mesh.cursor = 'pointer';
mesh.on('pointertap', () => spin(mesh));

  4. 04ContainerTexture

    2D on 3D

    Render any 2D container onto a mesh every frame.

    const face = new ContainerTexture({ container: ui });


cube.material = new Material3D({ baseColorTexture: face });
app.ticker.add(() => face.render(app.renderer));
